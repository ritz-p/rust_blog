## オブジェクトとプロシージャ

```nim
type
  Greeting = object
    name: string
    count: int

proc message(greeting: Greeting): string =
  "Hello, " & greeting.name & " <tag> & value"

let greetings = @[
  Greeting(name: "世界", count: 1),
  Greeting(name: "Nim", count: 2)
]

for greeting in greetings:
  if greeting.count > 0:
    echo message(greeting)
```
