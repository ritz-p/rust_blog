```sql
-- 日本語コメントと引用符
SELECT category, COUNT(*) AS total
FROM articles
WHERE title LIKE '%Rust%'
  AND author = 'O''Reilly'
GROUP BY category
HAVING COUNT(*) > 1
ORDER BY total DESC;
```
